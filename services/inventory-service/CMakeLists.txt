cmake_minimum_required(VERSION 3.16)
project(shopselfcheckout_inventory_service)
set(CMAKE_CXX_STANDARD 20)
find_package(cpprestsdk REQUIRED)
add_executable(inventory_service src/main.cpp src/inventory_manager.cpp)
target_link_libraries(inventory_service PRIVATE cpprestsdk::cpprest pthread)

include(CTest)
if(BUILD_TESTING)
    add_executable(inventory_manager_test tests/inventory_manager_test.cpp src/inventory_manager.cpp)
    add_test(NAME inventory_manager_test COMMAND inventory_manager_test)
endif()
